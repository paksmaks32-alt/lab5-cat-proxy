# Перевірка кешуючого проксі-сервера (ЛР5) за допомогою curl
# Сервер запущено: npm run dev  (node main.js -h 127.0.0.1 -p 3000 -c cache)
# Команди для Windows cmd; -s -D - -o nul показує лише заголовки, без двійкового тіла картинки

# --- Параметри командного рядка ---
# без обов'язкових опцій: повідомлення про помилку, код завершення 1
node main.js
echo %errorlevel%

# --- GET ---
# 1. картинки немає в кеші: 200, X-Cache-Source: http.cat, файл з'являється в cache/
curl.exe -s -D - -o nul http://127.0.0.1:3000/201
# 2. повторний запит: 200, X-Cache-Source: cache
curl.exe -s -D - -o nul http://127.0.0.1:3000/201
# 3. неіснуючий код: 404, у кеш нічого не зберігається
curl.exe -i http://127.0.0.1:3000/999
# 4. зберегти картинку у файл і переконатися, що це JPEG
curl.exe -s -o cat418.jpg http://127.0.0.1:3000/418

# --- PUT ---
# 5. записати власну картинку під кодом 500: 201
curl.exe -i -X PUT --data-binary @my.jpg http://127.0.0.1:3000/500
# 6. GET після PUT: повертається власна картинка, X-Cache-Source: cache
curl.exe -s -D - -o nul http://127.0.0.1:3000/500
# 7. PUT без тіла: 400
curl.exe -i -X PUT http://127.0.0.1:3000/501

# --- DELETE ---
# 8. видалити картинку з кешу: 200
curl.exe -i -X DELETE http://127.0.0.1:3000/500
# 9. повторне видалення: 404
curl.exe -i -X DELETE http://127.0.0.1:3000/500
# 10. GET після DELETE: знову з http.cat
curl.exe -s -D - -o nul http://127.0.0.1:3000/500

# --- Помилкові запити ---
# 11. недозволений метод: 405, заголовок Allow: GET, PUT, DELETE
curl.exe -i -X POST http://127.0.0.1:3000/200
# 12. некоректний шлях: 400
curl.exe -i http://127.0.0.1:3000/abc
